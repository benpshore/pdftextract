"""Bounded clean-binary rejection checks for the four preserved carrier defects."""
import hashlib
import json
import os
from pathlib import Path
import subprocess
import sys
import time

binary = Path(sys.argv[1]).resolve()
destination = Path(sys.argv[2]).resolve()
destination.mkdir(exist_ok=False)
sources = Path('/workspace/tpe-pdf-oxide-evidence-20261015/fixtures-flatten-carriers-baseline01')

def sha(path):
    return hashlib.sha256(path.read_bytes()).hexdigest()

receipt = {'scope':'Four supervised negative controls for concrete flatten carrier defects; no general visual or safety qualification',
           'source_commit':subprocess.check_output(['git','rev-parse','HEAD'],text=True).strip(),
           'source_tree':subprocess.check_output(['git','rev-parse','HEAD^{tree}'],text=True).strip(),
           'git_status_before':subprocess.check_output(['git','status','--porcelain'],text=True),
           'binary':str(binary),'binary_sha256':sha(binary),
           'binary_version':subprocess.check_output([str(binary),'--version'],text=True).strip(),
           'script_sha256':sha(Path(__file__)),'cases':[]}
assert not receipt['git_status_before']
for kind in ['annotation','form']:
    for shape, reason in [('no-contents','explicit /Contents'),('inherited-resources','local /Resources')]:
        name = f'flatten-{kind}-{shape}'
        source = sources/(name+'-source.pdf')
        before = sha(source)
        run = destination/name
        run.mkdir()
        captures = run/'captures'
        captures.mkdir()
        request = run/'request.json'
        request.write_text(json.dumps({'operation':'edit.annotations' if kind=='annotation' else 'forms.flatten',
                                       'inputs':[str(source)],'pages':[1,1] if kind=='annotation' else None,
                                       'params':{'action':'flatten'} if kind=='annotation' else {}})+'\n')
        request.chmod(0o600)
        command = ['prlimit','--as=2147483648','--core=0','--','timeout','-k','2','15',str(binary),
                   'oxide','--request',str(request),'--out',str(run/'out'),'--timeout-ms','5000',
                   '--max-memory-growth-mib','256','--max-output-bytes','1048576']
        started = time.monotonic()
        with (run/'response.json').open('wb') as output, (run/'stderr.txt').open('wb') as errors:
            result = subprocess.run(command,stdout=output,stderr=errors,timeout=20,check=False,
                                    env={**os.environ,'TMPDIR':str(captures),'TMP':str(captures)})
        value = json.loads((run/'response.json').read_bytes())
        checks = {'explicit_failure':result.returncode!=0 and value.get('status')=='failed',
                  'expected_carrier_reason':reason in value.get('error',''),
                  'source_unchanged':sha(source)==before,'no_publication':not (run/'out').exists(),
                  'captures_clean':not any(captures.iterdir()),
                  'no_success_artifact_payload':not value.get('output') and not value.get('output_paths')}
        receipt['cases'].append({'name':name,'source_sha256':before,'request_sha256':sha(request),
                                 'command':command,'exit_code':result.returncode,
                                 'elapsed_seconds':round(time.monotonic()-started,3),
                                 'response_sha256':sha(run/'response.json'),'stderr_sha256':sha(run/'stderr.txt'),
                                 'response':value,'checks':checks,'passed':all(checks.values())})
receipt['git_status_after'] = subprocess.check_output(['git','status','--porcelain'],text=True)
receipt['binary_unchanged'] = sha(binary)==receipt['binary_sha256']
receipt['passed'] = all(row['passed'] for row in receipt['cases']) and receipt['binary_unchanged'] and not receipt['git_status_after']
(destination/'receipt.json').write_text(json.dumps(receipt,indent=2)+'\n')
print(json.dumps({'passed':receipt['passed'],'cases':[{k:row[k] for k in ['name','passed','checks']} for row in receipt['cases']],
                  'receipt':str(destination/'receipt.json')}))
sys.exit(0 if receipt['passed'] else 1)
