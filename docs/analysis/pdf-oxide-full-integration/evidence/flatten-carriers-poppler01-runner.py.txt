"""Observe only the finite, approved synthetic flatten carrier counterexamples."""
import hashlib
import json
from pathlib import Path
import subprocess

root = Path('/workspace/tpe-pdf-oxide-evidence-20261015')
source_root = root/'fixtures-flatten-carriers-baseline01'
destination = root/'flatten-carriers-poppler01'
destination.mkdir(exist_ok=False)

def sha(path):
    return hashlib.sha256(path.read_bytes()).hexdigest()

rows = []
for source in sorted(source_root.glob('*.pdf')):
    before = sha(source)
    text = destination/(source.stem+'.txt')
    command = ['prlimit','--as=1073741824','--core=0','--','timeout','-k','2','10',
               'pdftotext','-f','1','-l','1','-layout',str(source),str(text)]
    result = subprocess.run(command,capture_output=True,timeout=15,check=False)
    prefix = destination/source.stem
    render = ['prlimit','--as=1073741824','--core=0','--','timeout','-k','2','15',
              'pdftoppm','-f','1','-singlefile','-r','96','-png',str(source),str(prefix)]
    rendered = subprocess.run(render,capture_output=True,timeout=20,check=False)
    image = prefix.with_suffix('.png')
    rows.append({'source':str(source),'source_sha256':before,'source_unchanged':sha(source)==before,
                 'text':{'command':command,'exit_code':result.returncode,
                         'stderr':result.stderr.decode(errors='replace'),
                         'output':text.read_text(errors='replace') if text.exists() else None,
                         'sha256':sha(text) if text.exists() else None},
                 'render':{'command':render,'exit_code':rendered.returncode,
                           'stderr':rendered.stderr.decode(errors='replace'),
                           'image':str(image),'sha256':sha(image) if image.exists() else None}})
receipt = {'scope':'Independent first-page render/text observations of synthetic flattening defects; no general visual or extraction qualification',
           'version':subprocess.run(['pdftotext','-v'],capture_output=True,timeout=5,check=False).stderr.decode(),
           'script_sha256':sha(Path(__file__)),'observations':rows}
(destination/'receipt.json').write_text(json.dumps(receipt,indent=2)+'\n')
print(json.dumps({'files':len(rows),'failed':[row['source'] for row in rows if not row['source_unchanged'] or row['text']['exit_code'] or row['render']['exit_code']],
                  'receipt':str(destination/'receipt.json')}))
